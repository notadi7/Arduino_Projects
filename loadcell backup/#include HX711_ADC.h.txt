#include <HX711_ADC.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Pins
#define LOADCELL_DOUT_PIN 3
#define LOADCELL_SCK_PIN 2
#define TRIG_PIN 4
#define ECHO_PIN 5
#define IR_SENSOR_GATE1 6  // Active LOW (0 = car detected)
#define IR_SENSOR_GATE2 7  // Active LOW (0 = car detected)
#define GATE1_IN1 8
#define GATE1_IN2 9
#define GATE2_IN1 10
#define GATE2_IN2 11

// Constants
const unsigned long GATE_OPEN_TIME = 3000; // 3 seconds operation
const float WEIGHT_MIN = 1.0;
const float WEIGHT_MAX = 50.0;
const float HEIGHT_MIN = 5.0;
const float HEIGHT_MAX = 11.0;
const unsigned long STABILIZATION_TIME = 1000; // 1s stabilization
const float WEIGHT_THRESHOLD = 0.5; // 0.5g change = stable
const float HEIGHT_THRESHOLD = 0.3; // 0.3cm change = stable

// Objects
HX711_ADC scale(LOADCELL_DOUT_PIN, LOADCELL_SCK_PIN);
LiquidCrystal_I2C lcd(0x27, 20, 4); // I2C address 0x27, 20x4 display

// System State
enum GateState { CLOSED, OPENING, OPEN, CLOSING };
GateState gate1State = CLOSED;
GateState gate2State = CLOSED;
unsigned long gateActionStartTime = 0;

// Measurement Stability
bool weightStable = false;
bool heightStable = false;
unsigned long weightStableTime = 0;
unsigned long heightStableTime = 0;
float previousWeight = 0;
float previousHeight = 0;

// Current Measurements
float currentWeight = 0;
float currentHeight = 0;

void setup() {
  Serial.begin(9600);
  
  // Initialize LCD
  lcd.init();
  lcd.backlight();
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("System Initializing");
  
  // Initialize Load Cell
  scale.begin();
  scale.start(2000);
  scale.setCalFactor(696.0); // Calibrate this!
  
  // Initialize Sensors
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(IR_SENSOR_GATE1, INPUT_PULLUP);
  pinMode(IR_SENSOR_GATE2, INPUT_PULLUP);
  
  // Initialize Motors
  pinMode(GATE1_IN1, OUTPUT);
  pinMode(GATE1_IN2, OUTPUT);
  pinMode(GATE2_IN1, OUTPUT);
  pinMode(GATE2_IN2, OUTPUT);
  
  // Close all gates
  stopGate1();
  stopGate2();
  
  delay(1000);
  lcd.clear();
}

void loop() {
  // 1. Take measurements
  takeMeasurements();
  
  // 2. Check stability for each measurement separately
  checkWeightStability();
  checkHeightStability();
  
  // 3. Update LCD display
  updateDisplay();
  
  // 4. Control gates only when both measurements are stable
  if (weightStable && heightStable) {
    controlGates();
  }
  
  delay(50);
}

void takeMeasurements() {
  // Get weight
  if (scale.update()) {
    currentWeight = abs(scale.getData());
  }
  
  // Get height
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  currentHeight = pulseIn(ECHO_PIN, HIGH) * 0.034 / 2;
}

void checkWeightStability() {
  if (abs(currentWeight - previousWeight) > WEIGHT_THRESHOLD) {
    weightStable = false;
    weightStableTime = millis();
  } else if (millis() - weightStableTime > STABILIZATION_TIME) {
    weightStable = true;
  }
  previousWeight = currentWeight;
}

void checkHeightStability() {
  if (abs(currentHeight - previousHeight) > HEIGHT_THRESHOLD) {
    heightStable = false;
    heightStableTime = millis();
  } else if (millis() - heightStableTime > STABILIZATION_TIME) {
    heightStable = true;
  }
  previousHeight = currentHeight;
}

void updateDisplay() {
  // Row 1: Weight and Height values
  lcd.setCursor(0, 0);
  lcd.print("W:");
  lcd.print(currentWeight, 1);
  lcd.print("g  H:");
  lcd.print(currentHeight, 1);
  lcd.print("cm");
  
  // Row 2: Separate stability indicators
  lcd.setCursor(0, 1);
  lcd.print("W:");
  lcd.print(weightStable ? "STABLE  " : "SETTLING");
  
  lcd.setCursor(10, 1);
  lcd.print("H:");
  lcd.print(heightStable ? "STABLE  " : "SETTLING");
  
  // Row 3: Gate status
  lcd.setCursor(0, 2);
  lcd.print("G1:");
  switch(gate1State) {
    case CLOSED: lcd.print("CLOSED "); break;
    case OPENING: lcd.print("OPENING"); break;
    case OPEN: lcd.print("OPEN   "); break;
    case CLOSING: lcd.print("CLOSING"); break;
  }
  
  lcd.setCursor(10, 2);
  lcd.print("G2:");
  switch(gate2State) {
    case CLOSED: lcd.print("CLOSED "); break;
    case OPENING: lcd.print("OPENING"); break;
    case OPEN: lcd.print("OPEN   "); break;
    case CLOSING: lcd.print("CLOSING"); break;
  }
  
  // Row 4: IR sensor status
  lcd.setCursor(0, 3);
  lcd.print("IR1:");
  lcd.print(digitalRead(IR_SENSOR_GATE1) == LOW ? "P " : "NP");
  
  lcd.setCursor(10, 3);
  lcd.print("IR2:");
  lcd.print(digitalRead(IR_SENSOR_GATE2) == LOW ? "P " : "NP");
}

void controlGates() {
  controlGate1();
  controlGate2();
}

// [Rest of the motor control functions remain exactly the same as previous version]
void controlGate1() {
  bool irDetected = (digitalRead(IR_SENSOR_GATE1) == LOW);
  
  switch(gate1State) {
    case CLOSED:
      if (currentWeight > WEIGHT_MIN && currentWeight < WEIGHT_MAX &&
          currentHeight > HEIGHT_MIN && currentHeight < HEIGHT_MAX) {
        startOpeningGate1();
      }
      break;
      
    case OPENING:
      if (millis() - gateActionStartTime >= GATE_OPEN_TIME) {
        finishOpeningGate1();
      }
      break;
      
    case OPEN:
      if (irDetected) {
        startClosingGate1();
      }
      break;
      
    case CLOSING:
      if (millis() - gateActionStartTime >= GATE_OPEN_TIME) {
        finishClosingGate1();
      }
      break;
  }
}

void controlGate2() {
  bool irDetected = (digitalRead(IR_SENSOR_GATE2) == LOW);
  
  switch(gate2State) {
    case CLOSED:
      if (currentWeight > WEIGHT_MIN && 
          (currentWeight >= WEIGHT_MAX || currentHeight <= HEIGHT_MIN || currentHeight >= HEIGHT_MAX)) {
        startOpeningGate2();
      }
      break;
      
    case OPENING:
      if (millis() - gateActionStartTime >= GATE_OPEN_TIME) {
        finishOpeningGate2();
      }
      break;
      
    case OPEN:
      if (irDetected) {
        startClosingGate2();
      }
      break;
      
    case CLOSING:
      if (millis() - gateActionStartTime >= GATE_OPEN_TIME) {
        finishClosingGate2();
      }
      break;
  }
}

void startOpeningGate1() {
  digitalWrite(GATE1_IN1, HIGH);
  digitalWrite(GATE1_IN2, LOW);
  gate1State = OPENING;
  gateActionStartTime = millis();
}

void finishOpeningGate1() {
  stopGate1();
  gate1State = OPEN;
}

void startClosingGate1() {
  digitalWrite(GATE1_IN1, LOW);
  digitalWrite(GATE1_IN2, HIGH);
  gate1State = CLOSING;
  gateActionStartTime = millis();
}

void finishClosingGate1() {
  stopGate1();
  gate1State = CLOSED;
}

void stopGate1() {
  digitalWrite(GATE1_IN1, LOW);
  digitalWrite(GATE1_IN2, LOW);
}

void startOpeningGate2() {
  digitalWrite(GATE2_IN1, HIGH);
  digitalWrite(GATE2_IN2, LOW);
  gate2State = OPENING;
  gateActionStartTime = millis();
}

void finishOpeningGate2() {
  stopGate2();
  gate2State = OPEN;
}

void startClosingGate2() {
  digitalWrite(GATE2_IN1, LOW);
  digitalWrite(GATE2_IN2, HIGH);
  gate2State = CLOSING;
  gateActionStartTime = millis();
}

void finishClosingGate2() {
  stopGate2();
  gate2State = CLOSED;
}

void stopGate2() {
  digitalWrite(GATE2_IN1, LOW);
  digitalWrite(GATE2_IN2, LOW);
}
